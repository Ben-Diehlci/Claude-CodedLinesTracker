  for (let i = 1; i < 80; i++) { const d = new Date(T0 - i * 86400000); if (d.getDay() % 6 && (i * 7919) % 5) days[`${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`] = c((i * 7919) % 900) }

    base: { life: c(17400), days, folded: [] },
  })
  const clock = mock.clock(on, { now: T0 })
  await $.session.start({ cwd: '/x/CodedLinesTracker', surface: 'desktop', isInteractive: true })
  await $.tool.call({ tool: 'Write', file_path: '/a/x.tsx', content: Array.from({ length: 300 }, (_, i) => i % 7 ? `const v${i} = ${i}` : '// c').join('\n') })
